def save(name,value):

def sign(message,source=None):
    key=SOURCES[source or message['sourceId']]['key']
    return dict(message=message,signature=hmac.new(key,m._scope.canonical_sha256(message).encode('ascii'),hashlib.sha256).hexdigest())
def receipts(e):
    out=[]
    for index,scan in enumerate(e['scans']):
        facts=[('observation',0,scan),('lease',0,scan['lease'])]
        facts.extend(('launcher',r['identityBefore']['pid'],r['launch']) for r in scan['processes'])
        for role,pid,payload in facts:
            out.append(sign(dict(sourceId=role,role=role,scanIndex=index,pid=pid,challenge=CHALLENGE,
                envelopeSha256=m._scope.canonical_sha256(e),subjectSha256=m._scope.canonical_sha256(e['subject']),
                factsSha256=m._scope.canonical_sha256(payload),observedAtMs=scan['endedAtMs'],expiresAtMs=115000)))
    return out

    v=verifier or m.EvidenceVerifier(fixture_sources=SOURCES)
    r=v.verify(e,rs,challenge=kw.get('challenge',CHALLENGE),now_ms=kw.get('now_ms',101000))
    assert e==before and rs==rb
    assert r['admissionDecision']=='unchanged' and r['executable'] is False and r['mayExempt'] is False
    assert r['realEvidenceAuthenticated'] is False and r['productionExecutable'] is False and r['compatibilityEnabled'] is False
    assert 'trust' not in r
    return r
